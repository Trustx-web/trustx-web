import { TrustXClient } from '@trustx/sdk';

const trustx = new TrustXClient({ apiKey: process.env.TRUSTX_SECRET_KEY });

export async function evaluateCheckout(orderData) {
  const decision = await trustx.decisions.evaluate({
    orderId: orderData.id,
    amount: orderData.total,
    currency: 'INR',
    customer: {
      email: orderData.email,
      phone: orderData.phone,
    },
    telemetryToken: orderData.sdkToken,
  });

  if (decision.action === 'BLOCK') {
    throw new Error('High risk transaction rejected by TrustX infrastructure');
  }

  return decision;
}
